{{- if eq .Values.strategy.type "canary" }}

1. Deploy New Version:
   helm upgrade {{ .Release.Name }} ./java-app-rollout \
     --set image.tag=NEW_VERSION \
     --reuse-values

2. Monitor Progress:
   kubectl argo rollouts get rollout {{ include "java-app-rollout.fullname" . }} --watch

3. Promote to Next Step:
   kubectl argo rollouts promote {{ include "java-app-rollout.fullname" . }}

4. Complete Rollout:
   kubectl argo rollouts promote {{ include "java-app-rollout.fullname" . }} --full

5. Rollback if Needed:
   kubectl argo rollouts abort {{ include "java-app-rollout.fullname" . }}
   kubectl argo rollouts undo {{ include "java-app-rollout.fullname" . }}

{{- else if eq .Values.strategy.type "blueGreen" }}

1. Deploy New Version:
   helm upgrade {{ .Release.Name }} ./java-app-rollout \
     --set image.tag=NEW_VERSION \
     --reuse-values

2. Test Preview Environment:
   kubectl port-forward svc/{{ include "java-app-rollout.fullname" . }}-preview 8081:80

3. Promote to Active:
   kubectl argo rollouts promote {{ include "java-app-rollout.fullname" . }}

4. Rollback if Needed:
   kubectl argo rollouts abort {{ include "java-app-rollout.fullname" . }}
   kubectl argo rollouts undo {{ include "java-app-rollout.fullname" . }}

{{- end }}
